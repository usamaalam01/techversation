```python
import requests

# Set API endpoint and API key
endpoint = "https://api.mistral.ai/ocr"
api_key = "YOUR_API_KEY"

# Set image file
image_file = "image.jpg"

# Send request to API

# Get recognized text
recognized_text = response.json()["text"]

print(recognized_text)
```
