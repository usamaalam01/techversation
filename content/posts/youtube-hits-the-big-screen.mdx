import sys

WIDTH, HEIGHT = 640, 480
WHITE = (255, 255, 255)

# Set up the display
screen = pygame.display.set_mode((WIDTH, HEIGHT))

# Game loop
while True:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            pygame.quit()
            sys.exit()

    # Fill the screen with white
    screen.fill(WHITE)

    # Draw a rectangle
    pygame.draw.rect(screen, (0, 0, 0), (100, 100, 200, 200))

    # Flip the display
    pygame.display.flip()
