# Example of differential privacy technique
import numpy as np

def add_noise(data, sensitivity, epsilon):
    """
    Add noise to data using the Laplace mechanism.
    
    Parameters:
    data (float): The original data value.
    sensitivity (float): The sensitivity of the query.
    epsilon (float): The privacy budget.
    
    Returns:
    float: The noisy data value.
    """
    noise = np.random.laplace(0, sensitivity / epsilon)
    return data + noise
