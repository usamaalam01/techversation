import google.auth
from googleapiclient.discovery import build

# Authenticate with Google API
creds = google.auth.default(scopes=['https://www.googleapis.com/auth/gmail.readonly'])

# Build the Gmail API client
gmail_service = build('gmail', 'v1', credentials=creds)

# Use Gemini Spark to summarize inbox
def summarize_inbox():
    # Get the inbox messages
    messages_response = gmail_service.users().messages().list(userId='me', labelIds=['INBOX']).execute()
    messages = messages_response.get('messages', [])

    # Summarize the messages using Gemini Spark
    summary = gemini_spark.summarize(messages)

    return summary
