# Example of a simple sleep tracker using Python
import datetime

class SleepTracker:
    def __init__(self):
        self.sleep_data = []

    def add_sleep_data(self, date, hours_slept):
        self.sleep_data.append((date, hours_slept))

    def get_average_sleep(self):
        total_hours = sum(hours for _, hours in self.sleep_data)
        return total_hours / len(self.sleep_data)

# Create a sleep tracker object

# Add some sleep data

tracker.add_sleep_data(datetime.date(2026, 5, 2), 8)
tracker.add_sleep_data(datetime.date(2026, 5, 3), 6)

# Get the average sleep
